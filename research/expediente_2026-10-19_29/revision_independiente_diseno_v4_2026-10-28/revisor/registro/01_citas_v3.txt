   750	            fcntl.flock(self.f.fileno(), fcntl.LOCK_EX | fcntl.LOCK_NB)

   758	        ev = {"utc": dt.datetime.now(dt.UTC).isoformat(), **ev}

   771	def permitir_intento(intentos: Intentos, sha_aut: str, texto: str):
   772	    """C11: hay una sola ejecución con resultado. Se rechaza, ANTES de crear la salida:
   773	      - si ya hay un EVALUADO o un INVALIDO;
   774	      - si hay un intento INICIADO sin estado terminal (el proceso murió o el disco falló: revisión manual);
   775	      - si el último ERROR ocurrió con estadísticos ya iniciados (reintentar sería una decisión metodológica);
   776	      - si, tras un ERROR previo a los estadísticos, la autorización es la misma o no cita el sha256 de ese RESULTADO."""
   777	    ev = intentos.eventos
   778	    terminales = [e for e in ev if e["evento"] in ("EVALUADO", "INVALIDO")]
   779	    if terminales:
   780	        raise Rechazo(f"E3 ya tiene un resultado terminal ({terminales[-1]['evento']}, {terminales[-1]['utc']}): no hay otra ejecución")
   781	    abiertos = ({e["id_intento"] for e in ev if e["evento"] == "INICIADO"}
   782	                - {e["id_intento"] for e in ev if e["evento"] in ("ERROR", "EVALUADO", "INVALIDO")})
   783	    if abiertos:
   784	        raise Rechazo(f"hay un intento sin estado terminal ({sorted(abiertos)}): requiere revisión manual antes de cualquier otro")
   785	    errores = [e for e in ev if e["evento"] == "ERROR"]
   786	    if not errores:
   787	        return None
   788	    u = errores[-1]
   789	    if u["estadisticos_iniciados"]:
   790	        raise Rechazo(f"el ERROR previo ocurrió en la etapa «{u['etapa']}», con estadísticos iniciados: "
   791	                      "reintentar es una decisión metodológica, no técnica")
   792	    if any(e.get("sha_autorizacion") == sha_aut for e in ev):
   793	        raise Rechazo("esta autorización ya se usó en un intento anterior: un reintento exige una autorización nueva")
   794	    if f"REINTENTO TRAS ERROR {u['sha_resultado']}" not in texto:
   795	        raise Rechazo(f"la autorización de reintento tiene que citar «REINTENTO TRAS ERROR {u['sha_resultado']}»")
   796	    return u
